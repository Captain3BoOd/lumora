# Vulkan backend: built when the Vulkan headers and loader are found (Linux CI installs them).
option(LUMORA_ENABLE_VULKAN "Build the Vulkan RHI backend when Vulkan is available" ON)
if(LUMORA_ENABLE_VULKAN)
  find_package(Vulkan QUIET)
  if(Vulkan_FOUND)
    target_sources(lumora_core PRIVATE rhi/vulkan/vulkan_device.cpp)
    if(NOT MSVC)
      # Vulkan structs are initialised with only sType set, the rest is zero by design.
      set_source_files_properties(rhi/vulkan/vulkan_device.cpp PROPERTIES COMPILE_OPTIONS -Wno-missing-field-initializers)
    endif()
    target_link_libraries(lumora_core PUBLIC Vulkan::Vulkan)
    target_compile_definitions(lumora_core PUBLIC LUMORA_HAS_VULKAN=1)
    message(STATUS "Lumora: Vulkan backend enabled")
  elseif(WIN32 AND LUMORA_WIN_VULKAN_SPIKE)
    include(WindowsVulkan)
    lumora_windows_vulkan_spike()
    if(LUMORA_WIN_VULKAN_OK)
      target_sources(lumora_core PRIVATE rhi/vulkan/vulkan_device.cpp)
      target_link_libraries(lumora_core PUBLIC lumora_vulkan_loader)
      target_compile_definitions(lumora_core PUBLIC LUMORA_HAS_VULKAN=1)
      message(STATUS "Lumora: Vulkan backend enabled (Windows spike, system loader)")
    endif()
  else()
    message(STATUS "Lumora: Vulkan not found, backend disabled")
  endif()
endif()

# GLES backend: Linux, needs EGL and GLESv2 (Mesa llvmpipe provides a software context in CI).
option(LUMORA_ENABLE_GLES "Build the GLES RHI backend when EGL and GLESv2 are available" ON)
if(LUMORA_ENABLE_GLES AND ANDROID)
  # NDK: EGL and GLESv3 are system libraries, the headers ship with the NDK sysroot.
  target_sources(lumora_core PRIVATE rhi/gles/gles_device.cpp)
  target_link_libraries(lumora_core PUBLIC EGL GLESv3)
  target_compile_definitions(lumora_core PUBLIC LUMORA_HAS_GLES=1)
  message(STATUS "Lumora: GLES backend enabled (Android)")
elseif(LUMORA_ENABLE_GLES AND UNIX AND NOT APPLE)
  find_library(LUMORA_EGL_LIB EGL)
  find_library(LUMORA_GLES_LIB GLESv2)
  find_path(LUMORA_GLES_INC GLES3/gl32.h)
  find_path(LUMORA_EGL_INC EGL/egl.h)
  if(LUMORA_EGL_LIB AND LUMORA_GLES_LIB AND LUMORA_GLES_INC AND LUMORA_EGL_INC)
    target_sources(lumora_core PRIVATE rhi/gles/gles_device.cpp)
    target_link_libraries(lumora_core PUBLIC ${LUMORA_EGL_LIB} ${LUMORA_GLES_LIB})
    target_compile_definitions(lumora_core PUBLIC LUMORA_HAS_GLES=1)
    message(STATUS "Lumora: GLES backend enabled")
  else()
    message(STATUS "Lumora: EGL or GLESv2 not found, GLES backend disabled")
  endif()
endif()
